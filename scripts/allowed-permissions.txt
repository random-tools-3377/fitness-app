# Every permission the release APK may ask for. Anything else fails CI.
# Network permissions are never allowed, whatever this file says.

# Added by AndroidX so the app's own broadcast receivers are private to it. Not a network permission.
${applicationId}.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION

# The rest timer between sets (0.2.0): an alarm on the phone, a notification and a buzz.
android.permission.POST_NOTIFICATIONS
android.permission.USE_EXACT_ALARM
android.permission.SCHEDULE_EXACT_ALARM
android.permission.VIBRATE

# Barcode scanning (0.3.0), asked for the first time the user scans. ZXing reads the image on the phone.
android.permission.CAMERA

# Reminders (0.4.0) are set again after the phone restarts.
android.permission.RECEIVE_BOOT_COMPLETED

# The privacy lock (0.4.0) asks for the phone's own fingerprint, face or PIN.
android.permission.USE_BIOMETRIC
android.permission.USE_FINGERPRINT

# Steps from the phone (0.10.0): read-only, from Health Connect, which works offline. Asked when
# the user turns steps on.
android.permission.health.READ_STEPS

# Saying a meal (0.18.0, optional AI): asked the first time the user taps Say it. The recording
# stays in memory, is turned into text on the phone, and is never saved.
android.permission.RECORD_AUDIO
